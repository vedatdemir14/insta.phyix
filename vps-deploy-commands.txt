# VPS Backend Deployment Komutları
# Bu komutları VPS terminalinde sırayla çalıştırın

# 1. Docker Compose'un çalışıp çalışmadığını kontrol et
docker compose version

# Eğer yukarıdaki komut çalışıyorsa, docker-compose yerine "docker compose" kullanın
# Eğer çalışmıyorsa, aşağıdaki komutla eski docker-compose'u kurun:

# Docker Compose kurulumu (eğer gerekirse)
DOCKER_COMPOSE_VERSION=$(curl -s https://api.github.com/repos/docker/compose/releases/latest | grep 'tag_name' | cut -d\" -f4)
curl -L "https://github.com/docker/compose/releases/download/${DOCKER_COMPOSE_VERSION}/docker-compose-$(uname -s)-$(uname -m)" -o /usr/local/bin/docker-compose
chmod +x /usr/local/bin/docker-compose
docker-compose --version

# 2. Proje dizini oluştur
mkdir -p /opt/instagram-scraper
cd /opt/instagram-scraper

# 3. Mevcut container'ları durdur
docker compose down 2>/dev/null || docker-compose down 2>/dev/null || true

# 4. Backend için docker-compose.yml oluştur
cat > docker-compose.yml << 'EOF'
version: '3.8'

services:
  backend:
    image: vedatdemir14/instagram-scraper-backend:latest
    container_name: instagram-scraper-backend
    ports:
      - "8000:8000"
    environment:
      - SUPABASE_URL=${SUPABASE_URL}
      - SUPABASE_API_KEY=${SUPABASE_API_KEY}
      - APIFY_API_TOKEN=${APIFY_API_TOKEN}
      - OPENROUTER_API_KEY=${OPENROUTER_API_KEY}
      - UNIPILE_API_KEY=${UNIPILE_API_KEY}
      - UNIPILE_BASE_URL=${UNIPILE_BASE_URL}
      - DEEPL_API_KEY=${DEEPL_API_KEY}
    restart: unless-stopped
    networks:
      - app-network
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:8000/health"]
      interval: 30s
      timeout: 10s
      retries: 3

networks:
  app-network:
    driver: bridge
EOF
